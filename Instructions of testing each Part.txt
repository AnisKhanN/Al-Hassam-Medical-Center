  -H "Authorization: Bearer <TOKEN>" \
  -d '{"patient":"<PATIENT_ID>","items":[{"description":"Consultation - Dr. Khan","category":"Consultation","unitPrice":1500},{"description":"Paracetamol x10","category":"Medicine","quantity":10,"unitPrice":15}]}'
# -> expect 201, billId "BIL-000001", subtotal 1650, totalAmount 1650, status "Unpaid"

# 2. Record a partial payment
curl -X POST http://localhost:5000/api/bills/<BILL_ID>/payments -H "Content-Type: application/json" \
  -H "Authorization: Bearer <TOKEN>" -d '{"amount":1000,"method":"Cash"}'
# -> expect balanceDue 650, status "Partially Paid"

# 3. Try to overpay
curl -X POST http://localhost:5000/api/bills/<BILL_ID>/payments -H "Content-Type: application/json" \
  -H "Authorization: Bearer <TOKEN>" -d '{"amount":9999,"method":"Cash"}'
# -> expect 400 "Payment exceeds balance due (Rs. 650)"

# 4. Pay the remainder
curl -X POST http://localhost:5000/api/bills/<BILL_ID>/payments -H "Content-Type: application/json" \
  -H "Authorization: Bearer <TOKEN>" -d '{"amount":650,"method":"Card","reference":"TXN-4521"}'
# -> expect status "Paid", balanceDue 0

# 5. Try cancelling a paid bill
curl -X PATCH http://localhost:5000/api/bills/<BILL_ID>/cancel -H "Authorization: Bearer <TOKEN>"
# -> expect 400 "Cannot cancel a bill with recorded payments..."

# 6. Create a second, unpaid bill and cancel it (should succeed — no payments yet)
curl -X PATCH http://localhost:5000/api/bills/<BILL_ID_2>/cancel -H "Authorization: Bearer <TOKEN>"

# 7. Search + filters
curl "http://localhost:5000/api/bills?status=Unpaid" -H "Authorization: Bearer <TOKEN>"
